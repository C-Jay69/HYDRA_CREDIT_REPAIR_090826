   196→function ProgressStepper({ currentStep }: { currentStep: number }) {
   197→  const steps = [
   198→    { num: 1, label: 'Upload', icon: Upload },
   199→    { num: 2, label: 'Review', icon: FileText },
   200→    { num: 3, label: 'Analyze', icon: Brain },
   201→  ];
   202→
   203→  return (
   204→    <div className="flex items-center justify-center gap-2">
   205→      {steps.map((step, i) => {
   206→        const isActive = currentStep === step.num;
   207→        const isCompleted = currentStep > step.num;
   208→        const Icon = step.icon;
   209→
   210→        return (
   211→          <div key={step.num} className="flex items-center gap-2">
   212→            {i > 0 && (
   213→              <div
   214→                className={cn(
   215→                  'h-0.5 w-8 md:w-16 transition-colors',
   216→                  currentStep > step.num ? 'bg-primary' : 'bg-muted'
   217→                )}
   218→              />
   219→            )}
   220→            <div className="flex flex-col items-center gap-1">
   221→              <div
   222→                className={cn(
   223→                  'flex size-10 items-center justify-center rounded-full border-2 transition-colors',
   224→                  isActive
   225→                    ? 'border-primary bg-primary text-primary-foreground'
   226→                    : isCompleted
   227→                      ? 'border-primary bg-primary/10 text-primary'
   228→                      : 'border-muted-foreground/30 bg-muted text-muted-foreground'
   229→                )}
   230→              >
   231→                {isCompleted ? (
   232→                  <CheckCircle2 className="size-5" />
   233→                ) : (
   234→                  <Icon className="size-4" />
   235→                )}
   236→              </div>
   237→              <span
   238→                className={cn(
   239→                  'hidden text-xs font-medium sm:block',
   240→                  isActive ? 'text-primary' : 'text-muted-foreground'
   241→                )}
   242→              >
   243→                {step.label}
   244→              </span>
   245→            </div>
   246→          </div>
   247→        );
   248→      })}
   249→    </div>
   250→  );
   251→}

   257→function Step1Upload({
   258→  items,
   259→  setItems,
   260→  reportSource,
   261→  setReportSource,
   262→  uploadedFile,
   263→  setUploadedFile,
   264→  onNext,
   265→}: {
   266→  items: ReportItem[];
   267→  setItems: React.Dispatch<React.SetStateAction<ReportItem[]>>;
   268→  reportSource: string;
   269→  setReportSource: (v: string) => void;
   270→  uploadedFile: string | null;
   271→  setUploadedFile: (v: string | null) => void;
   272→  onNext: () => void;
   273→}) {
   274→  const fileInputRef = useRef<HTMLInputElement>(null);
   275→  const [dragOver, setDragOver] = useState(false);
   276→
   277→  const addItem = () => {
   278→    setItems((prev) => [...prev, createEmptyItem()]);
   279→  };
   280→
   281→  const removeItem = (id: string) => {
   282→    setItems((prev) => prev.filter((item) => item.id !== id));
   283→  };
   284→
   285→  const updateItem = (id: string, field: keyof ReportItem, value: string | boolean) => {
   286→    setItems((prev) =>
   287→      prev.map((item) => (item.id === id ? { ...item, [field]: value } : item))
   288→    );
   289→  };
   290→
   291→  const handleDragOver = useCallback((e: React.DragEvent) => {
   292→    e.preventDefault();
   293→    setDragOver(true);
   294→  }, []);
   295→
   296→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   297→    e.preventDefault();
   298→    setDragOver(false);
   299→  }, []);
   300→
   301→  const handleDrop = useCallback((e: React.DragEvent) => {
   302→    e.preventDefault();
   303→    setDragOver(false);
   304→    const files = e.dataTransfer.files;
   305→    if (files.length > 0 && files[0].type === 'application/pdf') {
   306→      setUploadedFile(files[0].name);
   307→    }
   308→  }, []);
   309→
   310→  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   311→    const files = e.target.files;
   312→    if (files && files.length > 0) {
   313→      setUploadedFile(files[0].name);
   314→    }
   315→  };
   316→
   317→  const hasValidItems = items.some(
   318→    (item) => item.accountName.trim().length > 0
   319→  );
   320→
   321→  return (
   322→    <div className="space-y-6">
   323→      {/* Report Source */}
   324→      <div>
   325→        <Label className="text-sm font-medium">Report Source</Label>
   326→        <Select value={reportSource} onValueChange={setReportSource}>
   327→          <SelectTrigger className="mt-1.5 w-full">
   328→            <SelectValue placeholder="Select a credit bureau or source" />
   329→          </SelectTrigger>
   330→          <SelectContent>
   331→            {REPORT_SOURCES.map((src) => (
   332→              <SelectItem key={src.value} value={src.value}>
   333→                {src.label}
   334→              </SelectItem>
   335→            ))}
   336→          </SelectContent>
   337→        </Select>
   338→      </div>
   339→
   340→      <Separator />
   341→
   342→      {/* Drag & Drop Zone */}
   343→      <div>
   344→        <Label className="text-sm font-medium">Upload Credit Report (PDF)</Label>
   345→        <div
   346→          onDragOver={handleDragOver}
   347→          onDragLeave={handleDragLeave}
   348→          onDrop={handleDrop}
   349→          onClick={() => fileInputRef.current?.click()}
   350→          className={cn(
   351→            'mt-1.5 flex cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed p-8 transition-colors',
   352→            dragOver
   353→              ? 'border-primary bg-primary/5'
   354→              : 'border-muted-foreground/25 hover:border-primary/50 hover:bg-muted/50'
   355→          )}
   356→        >
   357→          <input
   358→            ref={fileInputRef}
   359→            type="file"
   360→            accept=".pdf"
   361→            className="hidden"
   362→            onChange={handleFileSelect}
   363→          />
   364→          {uploadedFile ? (
   365→            <>
   366→              <CheckCircle2 className="size-10 text-emerald-500" />
   367→              <p className="mt-3 text-sm font-medium">{uploadedFile}</p>
   368→              <p className="mt-1 text-xs text-muted-foreground">
   369→                PDF received. Now add the accounts from your report below so they can be analyzed.
   370→              </p>
   371→            </>
   372→          ) : (
   373→            <>
   374→              <Upload className="size-10 text-muted-foreground" />
   375→              <p className="mt-3 text-sm font-medium text-muted-foreground">
   376→                Drop your credit report here or click to browse
   377→              </p>
   378→              <p className="mt-1 text-xs text-muted-foreground">
   379→                PDF files only. Maximum 25MB.
   380→              </p>
   381→            </>
   382→          )}
   383→        </div>
   384→      </div>
   385→
   386→      <Separator />
   387→
   388→      {/* Manual Entry */}
   389→      <div>
   390→        <div className="flex items-center justify-between">
   391→          <div>
   392→            <Label className="text-sm font-medium">Manual Entry</Label>
   393→            <p className="text-xs text-muted-foreground">
   394→              Enter report items from your credit report manually
   395→            </p>
   396→          </div>
   397→          <Button size="sm" variant="outline" onClick={addItem}>
   398→            <Plus className="mr-1 size-4" />
   399→            Add Item
   400→          </Button>
   401→        </div>
   402→
   403→        {items.length === 0 ? (
   404→          <Card className="mt-3">
   405→            <CardContent className="flex flex-col items-center justify-center py-8 text-center">
   406→              <Inbox className="size-8 text-muted-foreground" />
   407→              <p className="mt-2 text-sm text-muted-foreground">
   408→                No items added yet. Click &quot;Add Item&quot; to begin.
   409→              </p>
   410→            </CardContent>
   411→          </Card>
   412→        ) : (
   413→          <ScrollArea className="mt-3 max-h-[500px]">
   414→            <div className="space-y-4">
   415→              {items.map((item, idx) => (
   416→                <Card key={item.id}>

   418→                    <div className="flex items-center justify-between">
   419→                      <CardTitle className="text-sm">
   420→                        Item #{idx + 1}

   422→                      {items.length > 1 && (
   423→                        <Button

   426→                          className="h-7 w-7 p-0 text-muted-foreground hover:text-destructive"

   428→                        >
   429→                          <Trash2 className="size-4" />
   430→                        </Button>
   431→                      )}
   432→                    </div>
   433→                  </CardHeader>

   435→                    <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   436→                      {/* Account Name */}

   439→                        <Input
