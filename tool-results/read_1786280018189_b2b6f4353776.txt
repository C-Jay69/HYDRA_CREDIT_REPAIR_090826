   603→function Step2Review({
   604→  items,
   605→  setItems,
   606→  jurisdiction,
   607→  setJurisdiction,
   608→  country,
   609→  setCountry,
   610→  onBack,
   611→  onAnalyze,
   612→  analyzing,
   613→}: {
   614→  items: ReportItem[];
   615→  setItems: React.Dispatch<React.SetStateAction<ReportItem[]>>;
   616→  jurisdiction: string;
   617→  setJurisdiction: (v: string) => void;
   618→  country: string;
   619→  setCountry: (v: string) => void;
   620→  onBack: () => void;
   621→  onAnalyze: () => void;
   622→  analyzing: boolean;
   623→}) {
   624→  const [editingId, setEditingId] = useState<string | null>(null);
   625→  const [editItem, setEditItem] = useState<ReportItem | null>(null);
   626→
   627→  const removeItem = (id: string) => {
   628→    setItems((prev) => prev.filter((item) => item.id !== id));
   629→  };
   630→
   631→  const startEdit = (item: ReportItem) => {
   632→    setEditingId(item.id);
   633→    setEditItem({ ...item });
   634→  };
   635→
   636→  const cancelEdit = () => {
   637→    setEditingId(null);
   638→    setEditItem(null);
   639→  };
   640→
   641→  const saveEdit = () => {
   642→    if (!editItem) return;
   643→    setItems((prev) =>
   644→      prev.map((item) => (item.id === editItem.id ? editItem : item))
   645→    );
   646→    setEditingId(null);
   647→    setEditItem(null);
   648→  };
   649→
   650→  const toggleFlag = (id: string, field: 'flaggedUnknown' | 'flaggedBalance') => {
