   101→function disputeStatusColor(status: string): string {
   102→  const s = status.toLowerCase();
   103→  if (s === 'resolved')
   104→    return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-300 dark:border-emerald-800';
   105→  if (s === 'denied')
   106→    return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800';
   107→  if (s === 'sent')
   108→    return 'bg-purple-100 text-purple-800 border-purple-200 dark:bg-purple-900/30 dark:text-purple-300 dark:border-purple-800';
   109→  if (s === 'in-progress')
   110→    return 'bg-blue-100 text-blue-800 border-blue-200 dark:bg-blue-900/30 dark:text-blue-300 dark:border-blue-800';
   111→  if (s === 'approved')
   112→    return 'bg-blue-100 text-blue-800 border-blue-200 dark:bg-blue-900/30 dark:text-blue-300 dark:border-blue-800';
   113→  if (s === 'pending-review')
   114→    return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800';
   115→  if (s === 'escalated')
   116→    return 'bg-orange-100 text-orange-800 border-orange-200 dark:bg-orange-900/30 dark:text-orange-300 dark:border-orange-800';
   117→  if (s === 'draft')
   118→    return 'bg-slate-100 text-slate-700 border-slate-200 dark:bg-slate-800/50 dark:text-slate-300 dark:border-slate-700';
   119→  return 'bg-slate-100 text-slate-700 border-slate-200 dark:bg-slate-800/50 dark:text-slate-300 dark:border-slate-700';
   120→}
   121→
   122→function disputeTypeColor(type: string): string {
   123→  const t = type.toLowerCase();
   124→  if (t.includes('outdated'))
   125→    return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800';
   126→  if (t.includes('duplicate'))
   127→    return 'bg-purple-100 text-purple-800 border-purple-200 dark:bg-purple-900/30 dark:text-purple-300 dark:border-purple-800';
   128→  if (t.includes('medical'))
   129→    return 'bg-pink-100 text-pink-800 border-pink-200 dark:bg-pink-900/30 dark:text-pink-300 dark:border-pink-800';
   130→  if (t.includes('sold') || t.includes('collection') || t.includes('fdcpa'))
   131→    return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800';
   132→  if (t.includes('authorized'))
   133→    return 'bg-blue-100 text-blue-800 border-blue-200 dark:bg-blue-900/30 dark:text-blue-300 dark:border-blue-800';
   134→  if (t.includes('double') || t.includes('charge-off'))
   135→    return 'bg-orange-100 text-orange-800 border-orange-200 dark:bg-orange-900/30 dark:text-orange-300 dark:border-orange-800';
   136→  if (t.includes('identity') || t.includes('theft'))
   137→    return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800';
   138→  return 'bg-slate-100 text-slate-700 border-slate-200 dark:bg-slate-800/50 dark:text-slate-300 dark:border-slate-700';
   139→}

   141→function confidenceBarColor(value: number): string {
   142→  if (value >= 75) return '[&>div]:bg-emerald-500';
   143→  if (value >= 50) return '[&>div]:bg-amber-500';
   144→  return '[&>div]:bg-red-500';
   145→}

   147→function formatStatus(status: string): string {
   148→  return status
   149→    .replace(/-/g, ' ')

   159→function formatDate(dateStr: string): string {
   160→  const d = new Date(dateStr);
   161→  return d.toLocaleDateString('en-US', {
   162→    month: 'short',
   163→    day: 'numeric',
   164→    year: 'numeric',
   165→  });
   166→}
   167→
   168→function formatShortDate(dateStr: string): string {
   169→  const d = new Date(dateStr);
   170→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   171→}

   242→function DisputeCard({
   243→  dispute,
   244→  onView,
   245→}: {
   246→  dispute: Dispute;
   247→  onView: () => void;
   248→}) {
   249→  const conf = dispute.confidence ?? 0;
   250→  const accountName = dispute.reportItem?.accountName ?? 'Unknown Account';
   251→  const creditorName =
   252→    dispute.reportItem?.creditorName ?? dispute.targetName ?? 'Unknown Creditor';
   253→
   254→  return (
   255→    <Card className="transition-shadow hover:shadow-md">
   256→      <CardContent className="p-5">
   257→        <div className="flex items-start justify-between gap-4">
   258→          {/* Left content */}
   259→          <div className="min-w-0 flex-1">
   260→            {/* Account name + creditor */}
   261→            <h3 className="truncate text-sm font-semibold">{accountName}</h3>
   262→            <p className="mt-0.5 truncate text-sm text-muted-foreground">
   263→              {creditorName}
   264→            </p>
   265→
   266→            {/* Badges row */}
   267→            <div className="mt-2 flex flex-wrap items-center gap-2">
   268→              <Badge
   269→                variant="outline"
   270→                className={cn(
   271→                  'text-[11px] capitalize',
   272→                  disputeTypeColor(dispute.disputeType)
   273→                )}
   274→              >
   275→                {formatDisputeType(dispute.disputeType)}
   276→              </Badge>
   277→              {dispute.strategy && (
   278→                <Badge
   279→                  variant="secondary"
   280→                  className="text-[11px] capitalize"
   281→                >
   282→                  {dispute.strategy.replace(/-/g, ' ')}
   283→                </Badge>
   284→              )}
   285→              <Badge
   286→                variant="outline"
   287→                className={cn(
   288→                  'text-[11px] capitalize',
   289→                  disputeStatusColor(dispute.status)
   290→                )}
   291→              >
   292→                {formatStatus(dispute.status)}
   293→              </Badge>
   294→            </div>
   295→
   296→            {/* Confidence bar */}
   297→            {conf > 0 && (
   298→              <div className="mt-3 flex items-center gap-2">
   299→                <Progress
   300→                  value={conf}
   301→                  className={cn('h-2 w-24', confidenceBarColor(conf))}
   302→                />
   303→                <span className="text-xs text-muted-foreground">{conf}%</span>
   304→              </div>
   305→            )}
   306→
   307→            {/* Legal basis */}
   308→            {dispute.legalBasis && (
   309→              <p className="mt-2 truncate text-xs text-muted-foreground italic">
   310→                <Scale className="mr-1 inline-block size-3" />
   311→                {dispute.legalBasis}
   312→              </p>
   313→            )}
   314→          </div>
   315→
   316→          {/* View button */}
   317→          <Button variant="outline" size="sm" onClick={onView} className="shrink-0">
   318→            <Eye className="mr-1.5 size-3.5" />
   319→            View
   320→          </Button>
   321→        </div>
   322→      </CardContent>
   323→    </Card>
   324→  );
   325→}

   331→function DisputeDetailDialog({
   332→  dispute,
   333→  open,
   334→  onOpenChange,
   335→}: {
   336→  dispute: Dispute | null;
   337→  open: boolean;
   338→  onOpenChange: (open: boolean) => void;
   339→}) {
   340→  const userId = useAppStore((s) => s.userId);
   341→  const userName = useAppStore((s) => s.userName);
   342→
   343→  const [selectedTemplate, setSelectedTemplate] = useState('');
   344→  const [letterPreview, setLetterPreview] = useState('');
   345→  const [disclaimerChecked, setDisclaimerChecked] = useState(false);
   346→  const [sending, setSending] = useState(false);
   347→  const [generating, setGenerating] = useState(false);
   348→
   349→  // Reset state when dialog opens with a new dispute
   350→  useEffect(() => {
   351→    if (open && dispute) {
   352→      setSelectedTemplate('');
   353→      setLetterPreview('');
   354→      setDisclaimerChecked(false);
   355→      setSending(false);
   356→      setGenerating(false);
   357→    }
   358→  }, [open, dispute]);
   359→
   360→  // Auto-select a recommended template based on dispute type
   361→  const recommendedTemplate = useMemo(() => {
   362→    if (!dispute) return '';
   363→    const dt = dispute.disputeType.toLowerCase();
   364→    if (dt.includes('fdcpa') || dt.includes('sold-debt') || dt.includes('collection'))
   365→      return 'fdcpa-809';
   366→    if (dt.includes('identity') || dt.includes('theft')) return 'identity-theft-605b';
   367→    if (dt.includes('medical')) return 'medical-debt-cfpb-2023';
   368→    if (dt.includes('duplicate')) return 'duplicate-reporting';
   369→    if (dt.includes('authorized')) return 'authorized-user-removal';
   370→    if (dt.includes('outdated') || dt.includes('double')) return 'fcra-611';
   371→    return 'fcra-611';
   372→  }, [dispute]);
   373→
   374→  const handleGenerateLetter = useCallback(() => {
   375→    if (!dispute || !selectedTemplate) return;
   376→
   377→    const template = TEMPLATE_CATALOG.find((t) => t.id === selectedTemplate);
   378→    if (!template) return;
   379→
   380→    setGenerating(true);
   381→
   382→    // Simulate async generation
   383→    setTimeout(() => {
   384→      try {
   385→        const accountName = dispute.reportItem?.accountName ?? 'Unknown Account';
   386→        const accountNumber = dispute.reportItem?.accountNumber ?? 'Unknown';
   387→        const creditorName =
   388→          dispute.reportItem?.creditorName ??
   389→          dispute.targetName ??
   390→          'Unknown Creditor';
   391→        const balance = dispute.reportItem?.balance ?? 0;
   392→
   393→        // Auto-detect bureau type from target name
   394→        let bureauInfo: { name: string; address: string } | null = null;
   395→        const targetLower = (dispute.targetName ?? '').toLowerCase();
   396→        if (targetLower.includes('experian'))
   397→          bureauInfo = BUREAU_ADDRESSES.experian;
   398→        else if (targetLower.includes('equifax'))
   399→          bureauInfo = BUREAU_ADDRESSES.equifax;
   400→        else if (targetLower.includes('transunion'))
   401→          bureauInfo = BUREAU_ADDRESSES.transunion;
   402→
   403→        const baseParams = {
   404→          userName: userName || 'Your Name',
   405→          userAddress: 'Your Address',
   406→          userCity: '',
   407→          userState: '',
   408→          userZip: '',
   409→          accountNumber,
   410→          creditorName,
   411→          amount: balance > 0 ? `$${balance.toFixed(2)}` : 'Unknown',
   412→          date: new Date().toISOString().split('T')[0],
   413→          reason: dispute.reason ?? `Dispute regarding ${accountName}`,
   414→        };
   415→
   416→        // Build template-specific params
   417→        let params: Record<string, unknown> = {
   418→          ...baseParams,
   419→          disputedItems: [accountName],
   420→          bureauName: bureauInfo?.name ?? 'Credit Bureau',
   421→          bureauAddress: bureauInfo?.address ?? 'P.O. Box 0000, City, State 00000',
   422→          collectorName: creditorName,
   423→          collectorAddress:
   424→            dispute.targetAddress ?? 'Unknown Address',
   425→          companyName: creditorName,
   426→          companyAddress:
   427→            dispute.targetAddress ?? 'Unknown Address',
   428→          disputeReason: dispute.reason ?? 'Inaccurate reporting',
   429→          product: 'Credit Reporting',
   430→          issue: `Inaccurate information regarding ${accountName}`,
   431→          desiredResolution: 'Deletion of inaccurate information',
   432→        };
   433→
   434→        const letter = template.templateFn(params);
   435→        setLetterPreview(letter);
   436→      } catch (err) {
   437→        console.error('Letter generation error:', err);
   438→        toast.error('Failed to generate letter. Check template parameters.');
   439→      } finally {
   440→        setGenerating(false);
   441→      }
   442→    }, 400);
   443→  }, [dispute, selectedTemplate, userName]);
   444→
   445→  const handleApproveAndSend = useCallback(async () => {
   446→    if (!dispute || !letterPreview || !selectedTemplate) return;
   447→
   448→    setSending(true);
   449→    try {
   450→      const template = TEMPLATE_CATALOG.find((t) => t.id === selectedTemplate);
   451→
   452→      // POST letter
   453→      const letterRes = await fetch('/api/letters', {
   454→        method: 'POST',
   455→        headers: { 'Content-Type': 'application/json' },
   456→        body: JSON.stringify({
   457→          disputeId: dispute.id,
   458→          templateType: template?.id ?? selectedTemplate,
   459→          recipientName: dispute.targetName,
   460→          recipientAddr: dispute.targetAddress,
   461→          content: letterPreview,
   462→        }),
   463→      });
   464→
   465→      if (!letterRes.ok) {
   466→        throw new Error('Failed to save letter');
   467→      }
   468→
   469→      // PUT dispute status to 'sent'
   470→ const disputeRes = await fetch('/api/disputes', {
   471→        method: 'PUT',
   472→        headers: { 'Content-Type': 'application/json' },
   473→        body: JSON.stringify({ id: dispute.id, status: 'sent' }),
   474→      });
   475→
   476→      if (!disputeRes.ok) {
   477→        throw new Error('Failed to update dispute status');
   478→      }
   479→
   480→      toast.success('Letter approved and dispute status updated to sent!');
   481→      onOpenChange(false);
   482→    } catch (err) {
   483→      console.error('Send error:', err);
   484→      toast.error('Failed to send letter. Please try again.');
   485→    } finally {
   486→      setSending(false);
   487→    }
   488→  }, [dispute, letterPreview, selectedTemplate, onOpenChange]);
   489→
   490→  const handleDownloadPdf = useCallback(() => {
   491→    toast.success('PDF ready for download');
   492→  }, []);
   493→
   494→  if (!dispute) return null;
   495→
   496→  const accountName = dispute.reportItem?.accountName ?? 'Unknown Account';
   497→  const creditorName =
   498→    dispute.reportItem?.creditorName ?? dispute.targetName ?? 'Unknown Creditor';
   499→  const conf = dispute.confidence ?? 0;
   500→
   501→  return (
   502→    <Dialog open={open} onOpenChange={onOpenChange}>
   503→      <DialogContent className="flex h-[85vh] max-w-5xl flex-col gap-0 p-0 sm:max-w-5xl">
   504→        <DialogHeader className="border-b px-6 py-4">
   505→          <DialogTitle className="flex items-center gap-2">
   506→            <FileText className="size-5 text-primary" />
   507→            {accountName}
   508→          </DialogTitle>
   509→          <DialogDescription>{creditorName}</DialogDescription>
   510→        </DialogHeader>
   511→
   512→        <div className="flex flex-1 overflow-hidden lg:flex-row">
   513→          {/* ---- Left Panel: Dispute Details ---- */}
   514→          <div className="flex w-full flex-col border-r lg:w-1/2">
   515→            <ScrollArea className="flex-1">
   516→              <div className="space-y-5 p-6">
   517→                {/* Status & confidence */}
   518→                <div>
   519→                  <h4 className="mb-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   520→                    Status
   521→                  </h4>
   522→                  <div className="flex flex-wrap items-center gap-2">
   523→                    <Badge
   524→                      variant="outline"
   525→                      className={cn(
   526→                        'text-xs capitalize',
   527→                        disputeStatusColor(dispute.status)
   528→                      )}
   529→                    >
   530→                      {formatStatus(dispute.status)}
   531→                    </Badge>
   532→                    {conf > 0 && (
   533→                      <div className="flex items-center gap-2">
   534→                        <Progress
   535→                          value={conf}
   536→                          className={cn(
   537→                            'h-2 w-16',
   538→                          confidenceBarColor(conf)
   539→                        )}
   540→                        />
   541→                        <span className="text-xs text-muted-foreground">
   542→                          {conf}%
   543→                        </span>
   544→                      </div>
   545→                    )}
   546→                  </div>
   547→                </div>
   548→
   549→                {/* Dispute info */}
   550→                <div>
   551→                  <h4 className="mb-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   552→                    Dispute Details
   553→                  </h4>
   554→                  <dl className="space-y-2 text-sm">
   555→                    <div className="flex justify-between">
   556→                      <dt className="text-muted-foreground">Type</dt>
   557→                      <dd className="font-medium">
   558→                        <Badge
   559→                          variant="outline"
   560→                          className={cn(
   561→                            'text-[11px] capitalize',
   562→                            disputeTypeColor(dispute.disputeType)
   563→                          )}
   564→                        >
   565→                          {formatDisputeType(dispute.disputeType)}
   566→                        </Badge>
   567→                      </dd>
   568→                    </div>
   569→                    {dispute.strategy && (
   570→                      <div className="flex justify-between">
   571→                        <dt className="text-muted-foreground">Strategy</dt>
   572→                        <dd className="font-medium capitalize">
   573→                          {dispute.strategy.replace(/-/g, ' ')}
   574→                        </dd>
   575→                      </div>

   928→export function DisputesPage() {
   929→  const userId = useAppStore((s) => s.userId);
   930→
   931→  const [disputes, setDisputes] = useState<Dispute[]>([]);
   932→  const [loading, setLoading] = useState(true);
   933→  const [error, setError] = useState<string | null>(null);
   934→  const [activeTab, setActiveTab] = useState('all');
   935→  const [typeFilter, setTypeFilter] = useState('all');
   936→  const [searchQuery, setSearchQuery] = useState('');
   937→  const [selectedDispute, setSelectedDispute] = useState<Dispute | null>(null);
   938→  const [dialogOpen, setDialogOpen] = useState(false);
   939→
   940→  // Fetch disputes
   941→  useEffect(() => {
   942→    if (!userId) return;
   943→    let cancelled = false;
   944→
   945→    async function fetchDisputes() {
   946→      try {
   947→        setLoading(true);
   948→        setError(null);
   949→        const res = await fetch(
   950→          `/api/disputes?userId=${encodeURIComponent(userId)}`
   951→        );
   952→        if (!res.ok) throw new Error('Failed to fetch disputes');
   953→        const json = await res.json();
   954→        if (!cancelled) {
   955→          setDisputes(json.data ?? []);
   956→        }
   957→      } catch (err) {
   958→        if (!cancelled) {
   959→          setError(
   960→            err instanceof Error ? err.message : 'An error occurred'
   961→          );
   962→        }
   963→      } finally {
   964→        if (!cancelled) setLoading(false);
   965→      }
   966→    }
   967→
   968→    fetchDisputes();
   969→    return () => {
   970→      cancelled = true;
   971→    };
   972→  }, [userId]);
   973→
   974→  // Unique dispute types for filter dropdown
   975→  const disputeTypes = useMemo(() => {
   976→    const types = new Set(disputes.map((d) => d.disputeType));
   977→    return Array.from(types).sort();

   979→
   980→  // Filtered disputes
   981→  const filteredDisputes = useMemo(() => {
   982→    let result = disputes;
   983→
   984→    // Tab filter
   985→    if (activeTab !== 'all') {
   986→      result = result.filter((d) => d.status === activeTab);
   987→    }
   988→
   989→    // Type filter
   990→    if (typeFilter !== 'all') {
   991→      result = result.filter((d) => d.disputeType === typeFilter);
   992→    }
   993→
   994→    // Search filter
   995→    if (searchQuery.trim()) {
   996→      const q = searchQuery.toLowerCase();
   997→      result = result.filter(
   998→        (d) =>
   999→          (d.reportItem?.accountName ?? '').toLowerCase().includes(q) ||
  1000→          (d.reportItem?.creditorName ?? '').toLowerCase().includes(q) ||
  1001→          (d.targetName ?? '').toLowerCase().includes(q)
  1002→      );
  1003→    }
  1004→
  1005→    return result;
  1006→  }, [disputes, activeTab, typeFilter, searchQuery]);
  1007→
  1008→  const handleViewDispute = useCallback((dispute: Dispute) => {
  1009→    setSelectedDispute(dispute);
  1010→    setDialogOpen(true);
  1011→  }, []);
  1012→
  1013→  // ---- Loading state ----
  1014→  if (loading) {
  1015→    return <DisputesSkeleton />;
  1016→  }
  1017→
  1018→  // ---- Error state ----

  1032→          >
  1033→            Retry

  1035→        </CardContent>
  1036→      </Card>
  1037→    );
  1038→  }
