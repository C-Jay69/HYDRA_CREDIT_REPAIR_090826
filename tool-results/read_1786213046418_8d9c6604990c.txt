    60→function daysUntil(dueDate: string): number {
    61→  const now = new Date();
    62→  now.setHours(0, 0, 0, 0);
    63→  const due = new Date(dueDate);
    64→  due.setHours(0, 0, 0, 0);
    65→  const diff = due.getTime() - now.getTime();
    66→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    67→}
    68→
    69→function formatCountdown(days: number): string {
    70→  if (days < 0) return `${Math.abs(days)} days overdue`;
    71→  if (days === 0) return 'Due today';
    72→  return `${days} day${days !== 1 ? 's' } remaining`;
    73→}
    74→
    75→function formatDate(dateStr: string): string {
    76→  const d = new Date(dateStr);
    77→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    78→}
    79→
    80→function formatShortDate(dateStr: string): string {
    81→  const d = new Date(dateStr);
    82→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
    83→}
    84→
    85→function deadlineTypeLabel(type: string): string {
    86→  if (type === '30-day-investigation') return '30-Day Investigation';
    87→  if (type === '5-day-validation') return '5-Day Validation';
    88→  if (type === 'sol-expiration') return 'SOL Expiration';
    89→  return type.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
    90→}
    91→
    92→function deadlineTypeColor(type: string): string {
    93→  const t = type.toLowerCase();
    94→  if (t.includes('30-day') || t.includes('investigation'))
    95→    return 'bg-blue-100 text-blue-800 border-blue-200 dark:bg-blue-900/30 dark:text-blue-300 dark:border-blue-800';
    96→  if (t.includes('5-day') || t.includes('validation'))
    97→    return 'bg-purple-100 text-purple-800 border-purple-200 dark:bg-purple-900/30 dark:text-purple-300 dark:border-purple-800';
    98→  if (t.includes('sol'))
    99→    return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800';
   100→  return 'bg-slate-100 text-slate-700 border-slate-200 dark:bg-slate-800/50 dark:text-slate-300 dark:border-slate-700';
   101→}
   102→
   103→function urgencyColor(days: number): string {
   104→  if (days < 0) return 'text-red-600 dark:text-red-400';
   105→  if (days < 7) return 'text-amber-600 dark:text-amber-400';
   106→  return 'text-emerald-600 dark:text-emerald-400';
   107→}
   108→
   109→function urgencyDotColor(days: number, isCompleted: boolean): string {
   110→  if (isCompleted) return 'bg-emerald-500';
   111→  if (days < 0) return 'bg-red-500';
   112→  if (days < 7) return 'bg-amber-500';
   113→  return 'bg-blue-500';
   114→}
   115→
   116→function urgencyLineColor(days: number, isCompleted: boolean): string {
   117→  if (isCompleted) return 'bg-emerald-200 dark:bg-emerald-800';
   118→  if (days < 0) return 'bg-red-200 dark:bg-red-800';
   119→  if (days < 7) return 'bg-amber-200 dark:bg-amber-800';
   120→  return 'bg-slate-200 dark:bg-slate-700';
   121→}
   122→
   123→function daysBadgeColor(days: number): string {
   124→  if (days < 0) return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800';
   125→  if (days === 0) return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800';
   126→  if (days < 7) return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800';
   127→  return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-300 dark:border-emerald-800';
   128→}

   495→// Deadlines Page
   496→// ---------------------------------------------------------------------------
   497→
   498→export function DeadlinesPage() {
   499→  const userId = useAppStore((s) => s.userId);
   500→
   501→  const [deadlines, setDeadlines] = useState<Deadline[]>([]);
   502→  const [loading, setLoading] = useState(true);
   503→  const [error, setError] = useState<string | null>(null);
   504→  const [viewMode, setViewMode] = useState<'timeline' | 'list'>('timeline');
   505→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc');
   506→  const [markingId, setMarkingId] = useState<string | null>(null);
   507→
   508→  // Fetch deadlines on mount
   509→  useEffect(() => {
   510→    if (!userId) return;
   511→    let cancelled = false;
   512→
   513→    async function fetchDeadlines() {
   514→      try {
   515→        setLoading(true);
   516→        setError(null);
   517→        const res = await fetch(`/api/deadlines?userId=${encodeURIComponent(userId)}`);
   518→        if (!res.ok) throw new Error('Failed to fetch deadlines');
   519→        const json = await res.json();
   520→        if (!cancelled) {
   521→          setDeadlines(json.data ?? []);
   522→        }
   523→      } catch (err) {
   524→        if (!cancelled) {
   525→          setError(err instanceof Error ? err.message : 'An error occurred');
   526→        }
   527→      } finally {
   528→        if (!cancelled) setLoading(false);
   529→      }
   530→    }
   531→
   532→    fetchDeadlines();
   533→    return () => { cancelled = true; };
   534→  }, [userId]);
   535→
   536→  // Mark deadline as complete
   537→  const handleMarkComplete = useCallback(async (id: string) => {
   538→    setMarkingId(id);
   539→    try {
   540→      const res = await fetch('/api/deadlines', {
   541→        method: 'PUT',
   542→        headers: { 'Content-Type': 'application/json' },
   543→        body: JSON.stringify({ id, isCompleted: true }),
   544→      });
   545→      if (!res.ok) throw new Error('Failed to update deadline');
   546→
   547→      // Update local state optimistically
   548→      setDeadlines((prev) =>
   549→        prev.map((d) => (d.id === id ? { ...d, isCompleted: true } : d))
   550→      );
   551→
   552→      toast.success('Deadline marked as complete!');
   553→    } catch (err) {
   554→      console.error('Mark complete error:', err);
   555→      toast.error('Failed to update deadline. Please try again.');
   556→    } finally {
   557→      setMarkingId(null);
   558→    }
   559→  }, []);
   560→
   561→  // Toggle sort direction
   562→  const handleToggleSort = useCallback(() => {
   563→    setSortDir((prev) => (prev === 'asc' ? 'desc' : 'asc'));
   564→  }, []);
   565→
   566→  // Sorted deadlines (by due date)
   567→  const sortedDeadlines = useMemo(() => {
   568→    return [...deadlines].sort((a, b) => {
   569→      const diff = new Date(a.dueDate).getTime() - new Date(b.dueDate).getTime();
   570→      return sortDir === 'asc' ? diff : -diff;
   571→    });
   572→  }, [deadlines, sortDir]);
   573→
   574→  // Summary stats
   575→  const summaryStats = useMemo(() => {
   576→    const now = new Date();
   577→    const weekFromNow = new Date(now);
   578→    weekFromNow.setDate(weekFromNow.getDate() + 7);
   579→
   580→    const total = deadlines.length;
   581→    const overdue = deadlines.filter((d) => !d.isCompleted && daysUntil(d.dueDate) < 0).length;
   582→    const dueThisWeek = deadlines.filter(
   583→      (d) => !d.isCompleted && daysUntil(d.dueDate) >= 0 && daysUntil(d.dueDate) <= 7
   584→    ).length;
   585→    const completed = deadlines.filter((d) => d.isCompleted).length;
   586→
   587→    return { total, overdue, dueThisWeek, completed };
   588→  }, [deadlines]);
   589→
   590→  // ---- Loading state ----
   591→  if (loading) {
   592→    return (
   593→      <div className="space-y-6">
   594→        <SummarySkeleton />
   595→        <TimelineSkeleton />
   596→      </div>
   597→    );
   598→  }
   599→
   600→  // ---- Error state ----
   601→  if (error) {
   602→    return (
   603→      <Card>
   604→        <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   605→          <div className="flex size-16 items-center justify-center rounded-full bg-red-100 dark:bg-red-900/30">
   606→            <AlertTriangle className="size-8 text-red-600 dark:text-red-400" />
   607→          </div>
   608→          <h3 className="mt-4 text-lg font-semibold">Error Loading Deadlines</h3>
   609→          <p className="mt-1 text-sm text-muted-foreground">{error}</p>
   610→          <Button variant="outline" className="mt-4" onClick={() => window.location.reload()}>
   611→            Retry
   612→          </Button>
   613→        </CardContent>
   614→      </Card>
   615→    );
   616→  }
   617→
   618→  return (
   619→    <div className="space-y-6">
   620→      {/* ---- Summary Cards ---- */}
   621→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   622→        <SummaryCard
   623→          label="Total Deadlines"
   624→          value={summaryStats.total}
   625→          icon={CalendarClock}
   626→          colorClass="text-primary"
   627→          iconBgClass="bg-primary/10"
   628→        />
   629→        <SummaryCard
   630→          label="Overdue"
   631→          value={summaryStats.overdue}
   632→          icon={AlertTriangle}
   633→          colorClass="text-red-600 dark:text-red-400"
   634→          iconBgClass="bg-red-100 dark:bg-red-900/30"
   635→        />
   636→        <SummaryCard
   637→          label="Due This Week"
   638→          value={summaryStats.dueThisWeek}
   639→          icon={Clock}
   640→          colorClass="text-amber-600 dark:text-amber-400"
   641→          iconBgClass="bg-amber-100 dark:bg-amber-900/30"
   642→        />
   643→        <SummaryCard
   644→          label="Completed"
   645→          value={summaryStats.completed}
   646→          icon={CheckCircle2}
   647→          colorClass="text-emerald-600 dark:text-emerald-400"
   648→          iconBgClass="bg-emerald-100 dark:bg-emerald-900/30"
   649→        />
   650→      </div>
   651→
   652→      {/* ---- View Toggle ---- */}
   653→      <div className="flex items-center justify-between">
   654→        <div className="flex items-center gap-2">
   655→          <h2 className="text-base font-semibold">All Deadlines</h2>
   656→          <Badge variant="secondary" className="text-[11px]">
   657→            {deadlines.length} total
   658→          </Badge>
   659→        </div>
   660→        <ToggleGroup
   661→          type="single"
   662→          value={viewMode}
   663→          onValueChange={(val) => {
   664→            if (val) setViewMode(val as 'timeline' | 'list');
   665→          }}
   666→          variant="outline"
   667→        >
   668→          <ToggleGroupItem value="timeline" className="gap-1.5 text-xs">
   669→            <GitBranch className="size-3.5" />
   670→            Timeline
   671→          </ToggleGroupItem>
   672→          <ToggleGroupItem value="list" className="gap-1.5 text-xs">
   673→            <List className="size-3.5" />
   674→            List
   675→          </ToggleGroupItem>
   676→        </ToggleGroup>
   677→      </div>
   678→
   679→      {/* ---- View Content ---- */}
   680→      {viewMode === 'timeline' ? (
   681→        <TimelineView
   682→          deadlines={sortedDeadlines}
   683→          onMarkComplete={handleMarkComplete}
   684→          markingId={markingId}
   685→        />
   686→      ) : (
   687→        <ListView
   688→          deadlines={sortedDeadlines}
   689→          onMarkComplete={handleMarkComplete}
   690→          markingId={markingId}
   691→          sortDir={sortDir}
   692→          onToggleSort={handleToggleSort}
   693→        />
   694→      )}
   695→    </div>
   696→  );
   697→}
