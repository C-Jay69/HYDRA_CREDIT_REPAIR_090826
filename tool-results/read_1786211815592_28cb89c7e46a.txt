   140→const LEGAL_DISCLAIMER =
   141→  'This communication is not legal advice. The sender is exercising their rights under applicable federal and state law.';
   142→
   143→function buildLetter(parts: {
   144→  recipientAddress?: string;
   145→  body: string;
   146→  recipientName?: string;
   147→  subject?: string;
   148→  referenceLine?: string;
   149→}): string {
   150→  const { recipientAddress, body, recipientName, subject, referenceLine } = parts;
   151→  const today = new Date().toLocaleDateString('en-US', {
   152→    year: 'numeric',
   153→    month: 'long',
   154→    day: 'numeric',
   155→  });
   156→  const lines: string[] = [];
   157→  lines.push(today);
   158→  lines.push('');
   159→  if (recipientName) lines.push(recipientName);
   160→  if (recipientAddress) lines.push(recipientAddress);
   161→  lines.push('');
   162→  if (subject) {
   163→    lines.push(`Re: ${subject}`);
   164→    lines.push('');
   165→  }
   166→  if (referenceLine) {
   167→    lines.push(referenceLine);
   168→    lines.push('');
   169→  }
   170→  lines.push(body.trim());
   171→  lines.push('');
   172→  lines.push(LEGAL_DISCLAIMER);
   173→  lines.push('');
   174→  return lines.join('\n');
   175→}
   176→
   177→function signatureBlock(userName: string, userAddress: string, userCity: string, userState: string, userZip: string): string {
   178→  return `Sincerely,\n\n${userName}\n${userAddress}\n${userCity}, ${userState} ${userZip}`;
   179→}

   185→export function generate609Letter(params: BureauDisputeParams): string {
   186→  const { userName, userAddress, userCity, userState, userZip, bureauName, bureauAddress } = params;
   187→
   188→  const body = `Dear ${bureauName},
   189→
   190→I am writing to request a complete copy of my consumer credit file, pursuant to my rights under the Fair Credit Reporting Act (FCRA), 15 U.S.C. \u00a7 1681g(a)(1) (Section 609).
   191→
   192→Under FCRA \u00a7 609(a)(1), every consumer reporting agency is required to disclose to a consumer, upon request, all information in the consumer's file at the time of the request. This includes, but is not limited to:
   193→
   194→  \u2022 All information in my credit file, including sources;
   195→  \u2022 All records of inquiries made regarding my file;
   196→  \u2022 The current credit score, if a score has been generated;
   197→  \u2022 A list of any persons or entities that have obtained a copy of my credit report within the past two years (or one year for employment purposes);
   198→  \u2022 All information regarding any items that have been the subject of a dispute.
   199→
   200→Please provide a complete copy of my consumer file within fifteen (15) days of receipt of this letter, as required under 15 U.S.C. \u00a7 1681i(a)(1).
   201→
   202→If you are unable to provide a complete disclosure, please identify in writing the specific information that is being withheld and the reason for such withholding.
   203→
   204→Please send all correspondence to the address listed below.
   205→
   206→${signatureBlock(userName, userAddress, userCity, userState, userZip)}`;
   207→
   208→  return buildLetter({
   209→    recipientName: bureauName,
   210→    recipientAddress: bureauAddress,
   211→    subject: `FCRA \u00a7 609 Request for Full File Disclosure \u2014 ${userName}`,
   212→    body,
   213→  });
   214→}

   217→// 2. FCRA \u00a7 611 — Dispute to Credit Bureaus
   218→// ---------------------------------------------------------------------------
   219→
   220→export function generate611Letter(params: BureauDisputeParams): string {
   221→  const {
   222→    userName, userAddress, userCity, userState, userZip,
   223→    bureauName, bureauAddress, reportReferenceNumber, disputedItems,
   224→  } = params;
   225→
   226→  const itemList = disputedItems.map((item, i) => `${i + 1}. ${item}`).join('\n');
   227→
   228→  const body = `Dear ${bureauName},
   229→
   230→I am writing to dispute the following information that appears on my consumer credit report. Pursuant to the Fair Credit Reporting Act (FCRA), 15 U.S.C. \u00a7 1681i(a)(1) (Section 611), I hereby request that you conduct a reasonable reinvestigation of the items listed below.
   231→
   232→Under FCRA \u00a7 611(a)(1), if the completeness or accuracy of any item of information contained in a consumer's file at a consumer reporting agency is disputed by the consumer and the consumer notifies the agency of such dispute, the agency shall conduct a reasonable reinvestigation to determine whether the disputed information is inaccurate.
   233→
   234→The following items are either inaccurate, incomplete, or unverifiable:
   235→
   236→${itemList}
   237→
   238→I am requesting that you:
   239→
   240→  1. Conduct a reasonable reinvestigation of each disputed item within thirty (30) days, as required by 15 U.S.C. \u00a7 1681i(a)(1);
   241→  2. Delete or modify the disputed information if it is found to be inaccurate, incomplete, or unverifiable;
   242→  3. Provide me with a corrected copy of my credit report upon completion of the reinvestigation;
   243→  4. Notify each furnisher of the disputed information that the information is disputed, as required by 15 U.S.C. \u00a7 1681i(a)(2).
   244→
   245→${reportReferenceNumber ? `Reference Number: ${reportReferenceNumber}\n` : ''}
   246→Please send all correspondence and results to the address listed below.
   247→
   248→${signatureBlock(userName, userAddress, userCity, userState, userZip)}`;
   249→
   250→  return buildLetter({
   251→    recipientName: bureauName,
   252→    recipientAddress: bureauAddress,
   253→    subject: `FCRA \u00a7 611 Dispute of Inaccurate Information \u2014 ${userName}`,
   254→    referenceLine: reportReferenceNumber ? `Reference Number: ${reportReferenceNumber}` : undefined,
   255→    body,
   256→  });
   257→}

   300→export function generateFdcpaValidation(params: DebtValidationParams): string {
   301→  const {
   302→    userName, userAddress, userCity, userState, userZip,
   303→    accountNumber, collectorName, collectorAddress, amount,
   304→    originalCreditor, noticeDate, collectorAccountNumber,
   305→  } = params;
   306→
   307→  const acct = collectorAccountNumber ?? accountNumber;
   308→
   309→  const body = `Dear ${collectorName},
   310→
   311→I am writing in response to your ${noticeDate ? `notice dated ${formatDateLong(noticeDate)}` : 'collection notice'} regarding the above-referenced account. Pursuant to my rights under the Fair Debt Collection Practices Act (FDCPA), 15 U.S.C. \u00a7 1692g(a)(3) (Section 809), I hereby request validation of this alleged debt.
   312→
   313→Under FDCPA \u00a7 809(a), within five (5) days after the initial communication with a consumer in connection with the collection of any debt, a debt collector shall send the consumer a written notice containing the amount of the debt, the name of the creditor, and a statement that the consumer has thirty (30) days to dispute the validity of the debt.
   314→
   315→I dispute the validity of all or part of this debt. Please provide the following verification:
   316→
   317→  \u2022 The original signed application or contract creating this alleged debt;
   318→  \u2022 A complete accounting and itemization of the amount claimed to be owed;
   319→  \u2022 Any judgment or court order related to this debt;
   320→  \u2022 Proof that your company has been assigned or authorized to collect this debt;
   321→  \u2022 The name and address of the original creditor (if different from your company);
   322→  \u2022 A copy of any assignment or transfer of this debt from the original creditor to your company;
   323→  \u2022 Proof that the statute of limitations has not expired on this debt.
   324→
   325→Account Information:
   326→  \u2022 Account Number: ${acct}
   327→  \u2022 Amount Claimed: ${amount}
   328→  ${originalCreditor ? `\u2022 Original Creditor: ${originalCreditor}` : ''}
   329→
   330→Please be advised that until you provide the requested validation, you may not continue to collect on this debt, and you must cease all collection activity, as required by 15 U.S.C. \u00a7 1692g(b). Please send all verification to the address listed below.
   331→
   332→${signatureBlock(userName, userAddress, userCity, userState, userZip)}`;
   333→
   334→  return buildLetter({
   335→    recipientName: collectorName,
   336→    recipientAddress: collectorAddress,
   337→    subject: `FDCPA \u00a7 809 Debt Validation Request \u2014 Account ${acct}`,
   338→    body,
   339→  });
   340→}

   343→// 5. FDCPA \u00a7 805(c) — Cease and Desist
   344→// ---------------------------------------------------------------------------
   345→
   346→export function generateCeaseDesist(params: CeaseDesistParams): string {
   347→  const {
   348→    userName, userAddress, userCity, userState, userZip,
   349→    accountNumber, collectorName, collectorAddress, amount,
   350→    prohibitPhoneCalls, prohibitWorkplaceContact,
   351→  } = params;
   352→
   353→  const prohibitions: string[] = [
   354→    'Immediately cease and desist all communication with me regarding the above-referenced debt.',
   355→  ];
   356→  if (prohibitPhoneCalls !== false) {
   357→    prohibitions.push('Do not contact me by telephone at any number, including my cell phone, home phone, or any other telephone number associated with me.');
   358→  }
   359→  if (prohibitWorkplaceContact !== false) {
   360→    prohibitions.push('Do not contact me at my place of employment or any workplace.');
   361→  }
   362→  prohibitions.push('You may only contact me to confirm that you will cease all collection efforts or to notify me of a specific remedy that you intend to pursue.');
   363→
   364→  const prohibitionList = prohibitions.map((p) => `  \u2022 ${p}`).join('\n');
   365→
   366→  const body = `Dear ${collectorName},
   367→
   368→I am exercising my right under the Fair Debt Collection Practices Act (FDCPA), 15 U.S.C. \u00a7 1692c(c) (Section 805(c)), to request that you immediately cease and desist all communication with me regarding the debt referenced below.
   369→
   370→Under FDCPA \u00a7 805(c), if a consumer notifies a debt collector in writing that the consumer refuses to pay a debt or that the consumer wishes the debt collector to cease further communication with the consumer, the debt collector shall not communicate further with the consumer with respect to such debt, except to advise the consumer that the debt collector's further efforts are being terminated or to notify the consumer that the debt collector or creditor may invoke specified remedies.
   371→
   372→Account Information:
   373→  \u2022 Account Number: ${accountNumber}
   374→  \u2022 Amount Claimed: ${amount}
   375→
   376→I am hereby directing you to:
   377→
   378→${prohibitionList}
   379→
   380→Please be advised that any further communication from you in violation of this request may constitute a violation of the FDCPA and may subject you to statutory damages, actual damages, and attorney's fees under 15 U.S.C. \u00a7 1692k.
   381→
   382→Please send all written correspondence regarding this matter to the address listed below.
   383→
   384→${signatureBlock(userName, userAddress, userCity, userState, userZip)}`;
   385→
   386→  return buildLetter({
   387→    recipientName: collectorName,
   388→    recipientAddress: collectorAddress,
   389→    subject: `FDCPA \u00a7 805(c) Cease and Desist \u2014 Account ${accountNumber}`,
   390→    body,
   391→  });
   392→}

   420→
   421→While I understand you are not legally obligated to make this adjustment, I am hopeful that you will review my overall positive account history and grant this request. Removing these late payments would significantly improve my credit standing and would reinforce my loyalty to your company.
   422→
   423→Please send your response to the address listed below.
   424→
   425→${signatureBlock(userName, userAddress, userCity, userState, userZip)}`;
   426→
   427→  return buildLetter({
   428→    recipientName: creditorName,
   429→    subject: `Goodwill Adjustment Request \u2014 Account ${accountNumber}`,

   431→  });

   486→export function generateIdentityTheftDispute(params: IdentityTheftParams): string {

   495→  const itemList = disputedItems?.map((item, i) => `${i + 1}. ${item}`).join('\n') ?? '  \u2022 Account Number: ' + accountNumber;
   496→
   497→  const body = `Dear ${creditorName},
   498→
   499→I am a victim of identity theft and am writing to dispute the following information that appears on my consumer credit report as a result of fraudulent activity. Pursuant to the Fair Credit Reporting Act (FCRA), 15 U.S.C. \u00a7 1681c-2 (Section 605B), I hereby request that you block the reporting of this fraudulent information.
   500→
   501→Identity Theft Report Information:

   503→  ${policeReportNumber ? `\u2022 Police Report Number: ${policeReportNumber}` : ''}
   504→  \u2022 Date Identity Theft Discovered: ${formatDateLong(discoveryDate)}
   505→
   506→Under FCRA \u00a7 605B(a)(1), a consumer reporting agency shall block the reporting of any information in the file of a consumer that the consumer identifies as resulting from identity theft, not later than four (4) business days after the date of receipt of the following information from the consumer:
   507→
   508→  1. The identity of the consumer;
   509→  2. A copy of an identity theft report;
   510→  3. The identification of the information in the consumer's file that is the result of identity theft;
   511→  4. A statement by the consumer that the information is not the result of any transaction by the consumer.
   512→
   513→${isFullBlock ? 'I am requesting a FULL BLOCK of all information in my credit file that is the result of identity theft.' : `The following specific items are the result of identity theft:\n\n${itemList}`}
   514→
   515→Account Information:
   516→  \u2022 Account Number: ${accountNumber}
   517→  \u2022 Amount: ${amount}
   518→  \u2022 Creditor: ${creditorName}
   519→
   520→I certify under penalty of perjury that the information identified above is not the result of any transaction by me and is the result of identity theft.
   521→
   522→Please block the reporting of this information within four (4) business days, as required by 15 U.S.C. \u00a7 1681c-2(a)(1), and provide me with written confirmation of the block.
   523→
